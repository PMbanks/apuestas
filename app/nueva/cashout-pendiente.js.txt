  /* =====================================================================================================
     4) CASH OUT (db/024_cashout.sql): retirar una apuesta abierta y cobrar ya su valor, con el margen de la casa.
     El valor lo calcula el servidor; aqui solo se pregunta (cada 20 s) y se pide, con confirmacion de dos toques.
     ===================================================================================================== */
  const CO = { t: 0, q: new Map(), vivo: false, off: 0, sure: null, busy: false };
  async function cargaCO(forzar) {
    if (CO.vivo || Date.now() < CO.off || (!forzar && Date.now() - CO.t < 20e3)) return;
    CO.vivo = true;
    try { const l = await rpc('cash_out_quotes'); CO.q = new Map((l || []).map((x) => [x.bet, x])); CO.t = Date.now() }
    catch (e) { CO.off = Date.now() + (noEsta(e) ? 30 : 2) * 60e3; CO.t = Date.now() }
    CO.vivo = false; rev();
  }
  function botonCO(id, qt) {
    const sure = CO.sure && CO.sure.bet === id, v = qt.value, dif = v - qt.stake;
    const sig = `${id}|${v}|${sure ? 1 : 0}`;
    const txt = sure ? `¿Cobrar? Toca otra vez <b>${N(v)}</b>${coin()}` : `Cash out <b>${N(v)}</b>${coin()}`;
    return { sig, html: `<button class="xc-btn ${sure ? 'sure' : dif < 0 ? 'neg' : ''}" data-xc="ask" data-bet="${id}" data-sig="${sig}" title="${dif >= 0 ? 'Ganas' : 'Pierdes'} ${N(Math.abs(dif))} frente a lo apostado">${txt}</button>` };
  }
  function parcheaCash() {
    if (CO.sure && Date.now() - CO.sure.t > 6000) CO.sure = null;
    // boletos de Mis apuestas: el numero de la apuesta esta en el propio boleto ("Nº 000123")
    document.querySelectorAll('.bol[data-goto]').forEach((art) => {
      const id = +((/(\d+)/.exec(art.querySelector('.bol-no')?.textContent || '') || [])[1] || 0), qt = CO.q.get(id), ya = art.querySelector('.xc-btn'), fila = art.querySelector('.bol-n');
      if (!fila || !id) return;
      if (!qt || !qt.ok) { if (ya) ya.remove(); return }
      const b = botonCO(id, qt);
      if (ya && ya.dataset.sig === b.sig) return;
      if (ya) ya.remove();
      fila.insertAdjacentHTML('beforeend', b.html); traduce(fila.lastElementChild);
    });
    // la tarjeta de tus apuestas en directo, encima del perfil: una por apuesta, en el mismo orden que las del servidor
    const porPartido = new Map();
    for (const qt of CO.q.values()) { const l = porPartido.get(qt.match) || []; l.push(qt); porPartido.set(qt.match, l) }
    const vistos = new Map();
    document.querySelectorAll('#vivoyo .vy[data-vivo]').forEach((vy) => {
      const m = vy.dataset.vivo, k = vistos.get(m) || 0; vistos.set(m, k + 1);
      const qt = (porPartido.get(m) || [])[k], ya = vy.nextElementSibling && vy.nextElementSibling.classList.contains('xc-vy') ? vy.nextElementSibling : null;
      if (!qt || !qt.ok) { if (ya) ya.remove(); return }
      const b = botonCO(qt.bet, qt);
      if (ya && ya.firstElementChild && ya.firstElementChild.dataset.sig === b.sig) return;
      if (ya) ya.remove();
      vy.insertAdjacentHTML('afterend', `<div class="xc-vy">${b.html}</div>`); traduce(vy.nextElementSibling);
    });
  }
  async function pideCO(id) {
    if (CO.busy) return;
    if (!(CO.sure && CO.sure.bet === id)) {
      // primer toque: se enseña el valor de ahora mismo y se pide confirmar
      CO.sure = { bet: id, t: Date.now() }; rev();
      try { const q = await rpc('cash_out_quote', { p_bet: id }); if (q && q.ok) CO.q.set(id, { ...CO.q.get(id), ...q, match: q.match || (CO.q.get(id) || {}).match }); else { CO.sure = null; avisa(esc((q && q.why) || 'Ahora no se puede retirar')); cargaCO(true) } }
      catch (e) { CO.sure = null; avisa(aviso(e)) }
      rev(); setTimeout(rev, 6200); return;
    }
    const qt = CO.q.get(id); CO.busy = true;
    try {
      const r = await rpc('cash_out', { p_bet: id, p_min: qt ? qt.value : null });
      avisa(`Cash out: cobrado <b>${N(r.value)}</b> · ${r.beneficio >= 0 ? 'ganas' : 'pierdes'} ${N(Math.abs(r.beneficio))}`);
      CO.q.delete(id); recarga();
    } catch (e) { avisa(aviso(e)) }
    CO.sure = null; CO.busy = false; await cargaCO(true); rev();
  }
